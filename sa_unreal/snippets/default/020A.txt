float x, y, z, radius = 10.0
Car carHandle

while true
    wait 100
    x, y, z = get_active_camera_coordinates
    if
        carHandle = get_random_car_in_sphere_no_save_recursive x y z radius findNext True skipWrecked True
    then
        lock_car_doors carHandle CarLock.LockoutPlayerOnly
    end
end
